\begin{frame}{Law of Total Probability \& Bayes' Rule}
    \textbf{Law of Total Probability}
    \begin{align*}
        P(A)                    & = \sum_{i} P(A \mid B_i) P(B_i)                           \\
        p(h_{t+1} \mid o_{1:t}) & = \int p(h_{t+1} \mid h_t) \, p(h_t \mid o_{1:t}) \, dh_t
    \end{align*}

    \vspace{0.4em}
    \hrule
    \vspace{0.6em}

    \textbf{Bayes' Rule}
    \begin{align*}
        P(A \mid B)               & = \frac{P(B \mid A) P(A)}{P(B)}                                                      \\
        p(h_{t+1} \mid o_{1:t+1}) & = \frac{p(o_{t+1} \mid h_{t+1}) \, p(h_{t+1} \mid o_{1:t})}{p(o_{t+1} \mid o_{1:t})}
    \end{align*}
\end{frame}

\begin{frame}{Recursive Bayesian Estimation}
    \centering
    % Transition arrow
    \begin{tikzpicture}[
        every node/.style={font=\normalsize},
        every edge/.style={draw, thick, -{Stealth[length=2.5mm]}}
        ]
        \node (curr) at (0, 0) {$b_t(h_t) := p(h_t \mid o_{1:t})$};
        \node (next) at (6.8, 0) {$b_{t+1}(h_{t+1}) := p(h_{t+1} \mid o_{1:t+1})$};

        \path (curr) edge node[above=0.15cm, font=\small\bfseries] {Recursive Bayesian Estimation} (next);
    \end{tikzpicture}

    \vspace{0.8em}

    \raggedright
    \textbf{1. Prediction Step} {\footnotesize (Law of Total Probability)}
    \[
        \overbrace{p(h_{t+1} \mid o_{1:t})}^{\text{Predicted Belief}} = \int \overbrace{\textcolor{blue!80!black}{p(h_{t+1} \mid h_t)}}^{\text{\textcolor{blue!80!black}{Transition Model}}} \, \underbrace{p(h_t \mid o_{1:t})}_{b_t(h_t)} \, dh_t
    \]

    \vspace{0.3em}

    \textbf{2. Update Step} {\footnotesize (Bayes' Rule)}
    \vspace{-1.8em}
    \begin{align*}
        b_{t+1}(h_{t+1}) = p(h_{t+1} \mid o_{1:t+1})
         & = \frac{\overbrace{\textcolor{teal!90!black}{p(o_{t+1} \mid h_{t+1})}}^{\text{\textcolor{teal!90!black}{Observation Model}}} \, p(h_{t+1} \mid o_{1:t})}{p(o_{t+1} \mid o_{1:t})} \\[0.5em]
         & = \frac{\textcolor{teal!90!black}{p(o_{t+1} \mid h_{t+1})} \, p(h_{t+1} \mid o_{1:t})}{\int \textcolor{teal!90!black}{p(o_{t+1} \mid h_{t+1})} \, p(h_{t+1} \mid o_{1:t}) \, dh_{t+1}}
    \end{align*}
\end{frame}

\begin{frame}{Belief Markov Chain}
    \centering
    \vspace{-0.4em}
    \textbf{$\{b_t\}$ is a Markov Chain:}
    \vspace{-0.7em}
    \[
        p(b_{t+1} \mid b_t, b_{t-1}, \dots, b_0) = p(b_{t+1} \mid b_t)
    \]

    \vspace{-0.2em}
    % Branching diagram
    \begin{tikzpicture}[
        every node/.style={font=\footnotesize},
        box/.style={draw, rounded corners, thick, inner sep=3.5pt, align=center, fill=gray!8},
        arrow/.style={draw, thick, -{Stealth[length=2.2mm]}}
        ]
        % Root: Current belief
        \node[box] (root) at (-2.2, 0) {$b_t = \begin{bmatrix} b_t(\text{Focused}) \\ b_t(\text{Distracted}) \end{bmatrix}$};

        % Branches
        \node[box, fill=teal!10] (up) at (6.6, 0.85) {$b_{t+1}^{(\text{Respond})}$};
        \node[box, fill=orange!10] (down) at (6.6, -0.85) {$b_{t+1}^{(\text{No Response})}$};

        % Edges with transition probabilities
        \path (root.east) edge[arrow] node[above=1pt, pos=0.52, sloped] {$o_{t+1} = \text{Respond} \quad p(\text{Respond} \mid b_t)$} (up.west);
        \path (root.east) edge[arrow] node[below=1pt, pos=0.52, sloped] {$o_{t+1} = \text{No Response} \quad p(\text{No-Resp} \mid b_t)$} (down.west);
    \end{tikzpicture}

    \vspace{0.4em}
    \hrule
    \vspace{0.4em}

    \[
        p(\text{Respond} \mid b_t) = \textcolor{teal!90!black}{0.9} \, p(h_{t+1} = \text{F} \mid b_t) + \textcolor{teal!90!black}{0.5} \, p(h_{t+1} = \text{D} \mid b_t)
    \]

    \vspace{0.2em}
    \raggedright
    \begin{itemize}
        \item Both branch probabilities and $b_{t+1}$ depend \textbf{only} on $b_t$.
        \item Observation history $o_{1:t}$ is \textbf{not needed}.
    \end{itemize}
\end{frame}

\begin{frame}{Partially Observable MDP (POMDP)}
    \centering
    \vspace{-0.2em}
    \textbf{POMDP = Partially Observable MDP}

    \vspace{0.6em}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.44\textwidth}
            \centering
            \textbf{MDP + Hidden Markov Chain}
            \vspace{0.4em}
            \begin{itemize}
                \item State $h_t$ is hidden.
                \item Hidden MC:
                \vspace{-0.3em}
                \[
                    h_{t+1} \sim p(h_{t+1} \mid h_t, a_t)
                \]
                \item $o_t$ is not Markovian:
                \vspace{-0.3em}
                \[
                    p(o_{t+1} \mid o_{1:t}) \ne p(o_{t+1} \mid o_t)
                \]
            \end{itemize}
        \end{column}
        \begin{column}{0.08\textwidth}
            \centering
            \vspace{5.5em}
            {\Large $\longrightarrow$}
        \end{column}
        \begin{column}{0.46\textwidth}
            \centering
            \textbf{MDP + Belief MC}
            \vspace{0.4em}
            \begin{itemize}
                \item Belief state $b_t(h_t) = p(h_t \mid o_{1:t})$.
                \item $\{b_t\}$ is a Markov Chain (RBE).
                \item MDP on beliefs:
                \vspace{-0.5em}
                \[
                    b_{t+1} = \tau(b_t, a_t, o_{t+1})
                \]
                \item Reward: $\rho(b_t, a_t) = \mathbb{E}_{h_t \sim b_t}[r_t(h_t, a_t)]$.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

            \end{itemize}

        \begin{column}{0.48\textwidth}
            \textbf{Observation ($o_t \in \{0, 1\}$)}
            \begin{itemize}
                \item $p(o_t = 1 \mid \text{Focused}) = 0.9$
                \item $p(o_t = 1 \mid \text{Distracted}) = 0.5$
                \item Unhired worker provides no observation
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.6em}
    \hrule
    \vspace{0.6em}

    \textbf{Belief MDP}
    \begin{itemize}
        \item State: continuous belief $b_t = (b_t^{(1)}, b_t^{(2)}) \in [0, 1]^2$.
        \item Bayes-optimal via Value Iteration:
        \[
            V^*(b) = \max_{a} \Big\{ \rho(b, a) + \gamma \sum_o p(o \mid b, a) \, V^*(\tau(b, a, o)) \Big\}
        \]
        \vspace{-1.5em}
        \item \textbf{Bottleneck}: coupled continuous DP is intractable for large $K$.
        \item \textbf{Gittins Index} (summer school): decouples into 1D indices for each worker!
    \end{itemize}
\end{frame}

\begin{frame}{Duality: Stochastic vs.\ Bayesian Bandits}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Stochastic (Frequentist)}
            \begin{itemize}
                \item Fixed environment $\nu \in \mathcal{E}$.
                \item Minimax regret:
                \vspace{-0.4em}
                \[
                    R_n^*(\mathcal{E}) = \inf_{\pi} \sup_{\nu \in \mathcal{E}} R_n(\pi, \nu)
                \]
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Bayesian}
            \begin{itemize}
                \item Prior $Q \in \Delta(\mathcal{E})$: $\nu \sim Q$.
                \item Bayesian regret:
                \vspace{-0.4em}
                \[
                    \mathrm{BR}_n(\pi, Q) = \mathbb{E}_{\nu \sim Q} [R_n(\pi, \nu)]
                \]
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.3em}
    \hrule
    \vspace{0.3em}

    \textbf{Minimax Duality via Sion's Theorem}
    \begin{itemize}
        \item Regret is convex in policy $\pi$ and linear in prior distribution $Q$.
        \item \textbf{Sion's minimax theorem} (compact $\mathcal{E}$) allows swapping $\inf$ and $\sup$:
        \vspace{-0.4em}
        \[
            \inf_{\pi} \sup_{\nu \in \mathcal{E}} R_n(\pi, \nu)
            = \sup_{Q \in \Delta(\mathcal{E})} \inf_{\pi} \mathrm{BR}_n(\pi, Q)
        \]
        \vspace{-0.6em}
        \item \textbf{Duality}: Minimax regret equals Bayes regret under worst-case prior $Q^*$.
        \item Frequentist minimax lower bounds can be proved via Bayesian priors!
    \end{itemize}

    \vspace{0.2em}
    {\tiny \textcolor{gray}{Reference: \cite{lattimore_bandit_2020}}}
\end{frame}

\begin{frame}{Imperfect-Information Games $\to$ Belief Games \& POMDPs}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Imperfect Info (e.g., Poker)}
            \begin{itemize}
                \item Hidden private states (cards).
                \item Actions leak private info; subgame solving must balance across hands.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Public Belief State (PBS)}
            \begin{itemize}
                \item Public history is common knowledge.
                \item Joint belief $\beta \in \Delta(\mathcal{S}_1 \times \mathcal{S}_2)$.
                \item Yields \textbf{perfect-information game} on belief states (Brown et al., ReBeL).
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.3em}
    \hrule
    \vspace{0.25em}

    \textbf{Reduction to POMDP (Known Opponent Strategy)}
    \begin{itemize}
        \item If opponent policy $\pi_{-i}$ is \textbf{fixed/known}, it is absorbed into dynamics:
        \vspace{-0.45em}
        \[
            p(s', o \mid s, a_i) = \sum_{a_{-i}} \pi_{-i}(a_{-i} \mid s, a_i) \, p(s', o \mid s, a_i, a_{-i})
        \]
        \vspace{-0.65em}
        \item Equilibrium collapses to a \textbf{single-agent POMDP} (Belief MDP).
    \end{itemize}

    \vspace{0.1em}
    {\tiny \textcolor{gray}{Reference: \cite{london_machine_learning_meetup_noam_2021}}}
\end{frame}

\begin{frame}{Bandits for Functions: From Discrete to Continuous}
    \small
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Discrete Multi-Armed Bandit}
            \begin{itemize}
                \item Action set $a \in \{1, \dots, K\}$.
                \item Independent arms: $y_t = \mu(a_t) + \epsilon_t$.
                \item Sampling arm $a$ gives \textbf{no information} about any arm $a' \neq a$.
                \item All arms can be evaluated in finite steps.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Arbitrary Bounded $f: \mathbb{R} \to [0, 1]$}
            \begin{itemize}
                \item Continuous domain $x \in \mathcal{X} \subset \mathbb{R}$.
                \item Any bounded function: $y_t = f(x_t) + \epsilon_t$.
                \item $\implies$ \textbf{$\infty$ independent arms}!
                \item Querying $f(x)$ reveals \textbf{zero information} about $f(x')$ for $x' \neq x$.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.25em}
    \hrule
    \vspace{0.2em}
    \pause
    \begin{itemize}
        \item There may exist a single spike arm $x^*$ with $f(x^*) = 1$, and $0$ elsewhere.
        \item No learning is possible without \textbf{constraining the information / complexity} of $f$.
    \end{itemize}
\end{frame}

\begin{frame}{Lipschitz Bandits: Discretization Reduction}
    \small
    \textbf{Reduction via Discretization: $\infty$ Arms $\to$ Finite Arms}
    \vspace{-0.4em}
    \begin{itemize}
        \item Lipschitz continuity $|f(x) - f(x')| \le L \|x - x'\|$ bounds variation in an $\epsilon$-ball.
        \item Covering the domain with $K_\epsilon \sim \epsilon^{-d}$ points reduces continuous space to a \textbf{finite discrete bandit}.
    \end{itemize}

    \vspace{0.05em}

\begin{frame}{Tractable Continuous Beliefs: Parametric Conjugacy}
    \small
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Infinite Dims vs.\ Parameters}
            \begin{itemize}
                \item On continuous spaces, general beliefs $b(s)$ are infinite-dimensional densities.
                \item Generic Bayes updates are intractable:
                $b_{t+1}(s) \propto p(o_{t+1} \mid s) \, b_t(s)$.
                \item \textbf{Parametric idea}: restrict belief to a family $b(s; \boldsymbol{\theta})$ with $\boldsymbol{\theta} \in \Theta \subset \mathbb{R}^m$.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Conjugacy $\implies$ Closed Updates}
            \begin{itemize}
                \item \textbf{Conjugacy}: prior $b_t \in \mathcal{P}_\Theta \implies$ posterior $b_{t+1} \in \mathcal{P}_\Theta$.
                \item Infinite update collapses to \textbf{finite algebraic update}:
                $\boldsymbol{\theta}_{t+1} = g(\boldsymbol{\theta}_t, a_t, o_{t+1})$.
                \item Belief state becomes a finite-dimensional vector $\boldsymbol{\theta}_t \in \Theta$.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.2em}
    \hrule
    \vspace{0.15em}

    \textbf{Canonical Form: Linear-Gaussian Conjugacy (Kalman Filter)}
    \begin{itemize}
        \item Joint distribution $(s, o)$ is multivariate Gaussian.
        \item \textbf{Conditioning keeps things Gaussian}: $s \mid o$ remains strictly Gaussian.
        \item This direct Bayesian conditioning step is precisely the \textbf{Kalman filter}.
        \item Belief update is exact and algebraic: only tracks mean $\boldsymbol{\mu}_t$ and covariance $\boldsymbol{\Sigma}_t$.
    \end{itemize}
\end{frame}
% ------------------------------------------------------------------------------
\begin{frame}{Random Functions via Gaussian Basis Expansion}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Linear Combination of Basis}
            \begin{itemize}
                \item Expand unknown function $f$ over basis $\boldsymbol{\phi}(x) = [\phi_1(x), \dots, \phi_M(x)]^\top$:
                \vspace{-0.3em}
                \[
                    f(x) = \sum_{m=1}^M w_m \phi_m(x) = \mathbf{w}^\top \boldsymbol{\phi}(x)
                \]
                \vspace{-0.5em}
                \item Basis: Fourier harmonics, Chebyshev polynomials $T_m(x)$, or splines.
                \item \textbf{Prior over weights}: $\mathbf{w} \sim \mathcal{N}(\mathbf{0}, \boldsymbol{\Sigma}_w)$.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Induced Gaussian Prior over $f$}
            \begin{itemize}
                \item Since linear transformations of Gaussians are Gaussian, for any $x, x'$:
                \vspace{-0.3em}
                \[
                    \mathbb{E}[f(x)] = \mathbb{E}[\mathbf{w}]^\top \boldsymbol{\phi}(x) = 0
                \]
                \vspace{-0.5em}
                \[
                    \mathrm{Cov}(f(x), f(x')) = \boldsymbol{\phi}(x)^\top \boldsymbol{\Sigma}_w \boldsymbol{\phi}(x')
                \]
                \vspace{-0.5em}
                \item Finite evaluations $[f(x_1), \dots, f(x_n)]^\top$ are \textbf{jointly Gaussian}.
            \end{itemize}
        \end{column}
    \end{columns}

\end{frame}

\begin{frame}{Recursive Bayesian Estimation: Function Observations}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{\textcolor{blue!80!black}{Prior}}
            \begin{itemize}
                \item Latent parameter: $h_t = \mathbf{w} \in \mathbb{R}^M$
                \item Prior on functions: $f(x) = \boldsymbol{\phi}(x)^\top \mathbf{w}$.
                \item Static hidden state: $\mathbf{w}_{t+1} = \mathbf{w}_t$.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{\textcolor{teal!90!black}{Observation Model}}
            \begin{itemize}
                \item Probe at $x_t \implies o_t = (x_t, y_t)$:
                \vspace{-0.7em}
                \[
                    y_t = f(x_t) + \varepsilon_t, \quad \varepsilon_t \sim \mathcal{N}(0, \sigma_n^2)
                \]
                \vspace{-1.5em}
                \item Likelihood: $p(o_t \mid \mathbf{w}) = \mathcal{N}(y_t; \boldsymbol{\phi}(x_t)^\top \mathbf{w}, \sigma_n^2)$.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.25em}
    \hrule
    \vspace{0.2em}

    \textbf{Conjugate Gaussian Belief Update (Kalman Filter)}
    \begin{itemize}
        \item \textbf{Conjugacy}: Prior $b_{t-1}(\mathbf{w}) = \mathcal{N}(\boldsymbol{\mu}_{t-1}, \boldsymbol{\Sigma}_{t-1})$ and linear Gaussian likelihood $\textcolor{teal!90!black}{p(o_t \mid \mathbf{w})}$ yield an exact Gaussian posterior:
        \vspace{-0.5em}
        \[
            b_t(\mathbf{w}) := p(\mathbf{w} \mid o_{1:t}) \propto \textcolor{teal!90!black}{p(o_t \mid \mathbf{w})} \, b_{t-1}(\mathbf{w}) = \mathcal{N}(\boldsymbol{\mu}_t, \boldsymbol{\Sigma}_t)
        \]
        \vspace{-1.8em}
        \item Closed-form recursive update is the standard \textbf{Kalman filter} ($\mathcal{O}(M^2)$ cost).
        \item \textbf{Function Posterior at $x^*$}: $f(x^*) \mid o_{1:t} \sim \mathcal{N}\big(\boldsymbol{\phi}(x^*)^\top \boldsymbol{\mu}_t, \, \boldsymbol{\phi}(x^*)^\top \boldsymbol{\Sigma}_t \boldsymbol{\phi}(x^*)\big)$.
    \end{itemize}
\end{frame}

\begin{frame}[t]{Bayesian Optimization: A POMDP over Function Space}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{POMDP Elements}
            \begin{itemize}
                \item \textbf{State} $s$: Unknown function $f$.
                \item \textbf{Action} $a_t$: Query point $x_t \in \mathcal{X}$.
                \item \textbf{Obs} $o_t$: $y_t = f(x_t) + \varepsilon_t$.
                \item \textbf{Belief} $b_t$: Distribution over $f$.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \uncover<2->{
                \textbf{Representing $b_t(f)$}
                \begin{itemize}
                    \item \textbf{Basis expansion} (weight space):\\
                    $f(x) = \boldsymbol{\phi}(x)^\top \mathbf{w}$, update via Kalman.
                    \item \textbf{Kernel trick} (function space):\\
                    Implicit infinite basis $\to$ \textbf{GP}.
                \end{itemize}
            }
        \end{column}
    \end{columns}

    \uncover<3->{
        \vspace{0.8em}
        \hrule
        \vspace{0.8em}

        \begin{columns}[T,onlytextwidth]
            \begin{column}{0.48\textwidth}
                \textbf{Simple Regret (Pure Exploration)}
                \vspace{-0.3em}
                \[
                    S_T = \max\nolimits_{x \in \mathcal{X}} f(x) - f(\hat{x}_T)
                \]
                \vspace{-0.5em}
                \begin{itemize}
                    \item Payoff only at $\hat{x}_T$.
                    \item E.g., hyperparameter tuning.
                \end{itemize}
            \end{column}

            \begin{column}{0.48\textwidth}
                \uncover<4->{
                    \textbf{Cumulative Regret (Bandit)}
                    \vspace{-0.3em}
                    \[
                        R_T = \sum\nolimits_{t=1}^T \big(\max\nolimits_x f(x) - f(x_t)\big)
                    \]
                    \vspace{-0.5em}
                    \begin{itemize}
                        \item Exploit during optimization.
                        \item E.g., online routing, clinical trials.
                    \end{itemize}
                }
            \end{column}
        \end{columns}
    }
\end{frame}

\begin{frame}{Gaussian Processes}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Definition: $f \sim \mathcal{GP}(m, k)$}
            \begin{itemize}
                \item Collection of random variables, any finite subset is jointly Gaussian:
                \vspace{-0.3em}
                \[
                    [f(x_1), \dots, f(x_n)]^\top \sim \mathcal{N}(\mathbf{m}, \mathbf{K})
                \]
                \vspace{-0.5em}
                \item Mean: $m(x) = \mathbb{E}[f(x)]$ (often $0$).
                \item Covariance: $\mathbf{K}_{ij} = k(x_i, x_j)$.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Covariance Kernel $k(x, x')$}
            \begin{itemize}
                \item Encodes spatial correlation:
                \vspace{-0.3em}
                \[
                    k(x, x') = \mathrm{Cov}(f(x), f(x'))
                \]
                \vspace{-0.5em}
                \item \textbf{Squared Exponential (RBF)}:
                \vspace{-0.3em}
                \[
                    k(x, x') = \sigma_f^2 \exp\Big(-\frac{\|x - x'\|^2}{2\ell^2}\Big)
                \]
                \vspace{-0.5em}
                \item Lengthscale $\ell$: smoothness / reach.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.3em}
    \hrule
    \vspace{0.25em}

    \textbf{Nonparametric Prior}: Instead of parameterizing $f(x; \mathbf{w})$, a GP directly specifies a consistent prior over the infinite-dimensional function space.
\end{frame}
