\begin{frame}{Law of Total Probability \& Bayes' Rule}
    \textbf{Law of Total Probability}
    \begin{align*}
        P(A)                  & = \sum_{i} P(A \mid B_i) P(B_i)                                     \\
        p(h_t \mid o_{1:t-1}) & = \int p(h_t \mid h_{t-1}) \, p(h_{t-1} \mid o_{1:t-1}) \, dh_{t-1}
    \end{align*}

    \vspace{0.4em}
    \hrule
    \vspace{0.6em}

    \textbf{Bayes' Rule}
    \begin{align*}
        P(A \mid B)         & = \frac{P(B \mid A) P(A)}{P(B)}                                          \\
        p(h_t \mid o_{1:t}) & = \frac{p(o_t \mid h_t) \, p(h_t \mid o_{1:t-1})}{p(o_t \mid o_{1:t-1})}
    \end{align*}
\end{frame}
